No 2 Membuat Beberapa Tabel

// Tabel users \\
CREATE TABLE users (
    id INTEGER AUTO_INCREMENT PRIMARY KEY,
    NAME  VARCHAR(255),
    email VARCHAR(255),
    PASSWORD VARCHAR(255)
);

// Tabel categories \\
CREATE TABLE categories (
    id INTEGER AUTO_INCREMENT PRIMARY KEY,
    name VARCHAR(255)
);

// Tabel items \\
CREATE TABLE items (
    id INTEGER AUTO_INCREMENT PRIMARY KEY,
    name VARCHAR(255),
    description VARCHAR(255),
    price INTEGER,
    stock INTEGER,
    category_id INTEGER,
    FOREIGN KEY (category_id) REFERENCES categories(id)
);

Soal No 4 Mengambil data dari database

// a. Mengambil data users kecuali kolom password \\
SELECT id, name, email FROM users;

// b1. Mengambil data items dengan harga di atas 1000000
SELECT * FROM items WHERE price > 1000000;

// b2. Mengambil data items dengan nama item 'watch' \\ 
SELECT * FROM items WHERE name LIKE '%watch%';

// c. Menampilkan data items join dengan kategori
SELECT items.name AS item_name, items.description, categories.name AS category_name
FROM items
JOIN categories ON items.category_id = categories.id;
